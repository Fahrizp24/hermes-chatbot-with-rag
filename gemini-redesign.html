package auth

import (
	"sync"
	"time"
)

type cachedClaims struct {
	claims    map[string]any
	expiresAt time.Time
}

type JWTCache struct {
	mu       sync.RWMutex
	entries  map[string]cachedClaims
	ttl      time.Duration
	maxSize  int
}

func NewJWTCache(ttl time.Duration, maxSize int) *JWTCache {
	return &JWTCache{
		entries: make(map[string]cachedClaims),
		ttl:     ttl,
		maxSize: maxSize,
	}
}

func (c *JWTCache) Get(tokenHash string) (map[string]any, bool) {
	c.mu.RLock()
	entry, ok := c.entries[tokenHash]
	c.mu.RUnlock()
	if !ok || time.Now().After(entry.expiresAt) {
		return nil, false
	}
	return entry.claims, true
}

func (c *JWTCache) Set(tokenHash string, claims map[string]any) {
	c.mu.Lock()
	defer c.mu.Unlock()
	if len(c.entries) >= c.maxSize {
		for k := range c.entries {
			delete(c.entries, k)
			break
		}
	}
	c.entries[tokenHash] = cachedClaims{
		claims:    claims,
		expiresAt: time.Now().Add(c.ttl),
	}
}
